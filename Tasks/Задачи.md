Пусть процесс с `PID` `P1` породил два сыновьих процесса с `PID` -ами `P2` и `P3` : 

```console
P1
├── P2
└── P3
```

```c
int main(int argc, char **argv) //PID = P1
{
	if (fork() == 0) { //PID = P2
		printf ("%d %d \n", getppid(), getpid());
		exit(0);
	}
	if (fork() == 0) { //PID = P3
		printf ("%d \n", getpid());
		exit(0);
	}
	
	return 0;
}
```

## Очень последовательные блоки

```c
#include <stdio.h>

int shared_var = 0;

    printf("T%d", shared_var);

    pthread_mutex_unlock(&mutex);

    for(int i=0; i<3; i++) {
        pthread_join(threads[i], NULL);
    }

    printf("U%d\n", shared_var);
    return 0;
}
```

## Вложения в работу

Гарантируются атомарность операций, отсутствие буферизации и успешные системные вызовы. Написать все возможные выводы процесса с множеством нитей.

```c
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

char a = 'a';

int thread_function2(void *arg) {
    int method = *(int *) arg;

    if (method == 0) {
        printf("%d", method + (int) pthread_self());
        return a + 1;
    } else {
        printf("%d", method + (int) pthread_self());
        char local_a = a + 1;
        pthread_exit(&local_a);
    }
}


void *thread_function1(void *arg) {
    int method = *(int *) arg;

    if (method == 0) {
        printf("%c%lu", a, pthread_self());
        char tmp = (char) thread_function2(arg);
        printf("%c%lu\n", tmp, pthread_self());
        return NULL;
    } else if (method == 1) {
        printf("%c%lu", a, pthread_self());
        char tmp = (char) thread_function2(arg);
        printf("%c%lu\n", tmp, pthread_self());
        pthread_exit(&method);
    } else {
        int error = 1;
        pthread_exit((void *) &error);
    }
}

int main(void) {
    pthread_t tid[2]; //TID=[A,B]
    int methods[2] = {0, 1};
    int result;

    for (int i = 0; i < 2; i++) {
        result = pthread_create(&tid[i], NULL, thread_function1, &methods[i]);
        if (result != 0) {
            exit(EXIT_FAILURE);
        }
    }

    for (int i = 0; i < 2; i++) {
        pthread_join(tid[i], NULL);
    }

    return 0;
}
```

## Простая арифметика

Гарантируются атомарность операций, отсутствие буферизации и успешные системные вызовы. Написать все возможные выводы процесса с множеством нитей.

```c
#include <stdio.h>
#include <pthread.h>
#include <stdlib.h>

int shared_value = 0;
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

void *thread_func(void *arg) {
    int tid = *(int *)arg;
    pthread_mutex_lock(&mutex);
    printf("%d: %d\n", tid, shared_value);
    shared_value += tid;
    pthread_mutex_unlock(&mutex);
    pthread_exit(NULL);
}

int main() {
    pthread_t threads[3];
    int tids[3]; //1,2,3

    for(int i = 0; i < 3; i++) {
        pthread_create(&threads[i], NULL, thread_func, &tids[i]);
```
